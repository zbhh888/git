#include<iostream>
using namespace std;

int main(){
    char num[1001];
    int count[10]={0};
    cin>>num;
    for(int i=0;num[i]!='\0';i++){
         count[num[i]-'0']++;   
    }
    for(int i=0;i<10;i++){
        if(count[i]>0){
            cout<<i<<":"<<count[i]<<endl;
        }
    }
    return 0;
}
