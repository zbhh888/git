#include<iostream>
using namespace std;
struct student{
    char id[17];
    int test;
    int exam;
};
int main(){
    int n;
    cin>>n;
    struct student stu[1001];
    for(int i=0;i<n;i++){
        cin>>stu[i].id>>stu[i].test>>stu[i].exam;
    }
    int m;
    cin>>m;
    for(int i=0;i<m;i++){
        int x;
        cin>>x;
        for(int j=0;j<n;j++){
            if(x==stu[j].test)
                cout<<stu[j].id<<" "<<stu[j].exam<<endl;
        }
    }
    return 0;
}
