#include<iostream>
using namespace std;
int main(){
    int n,k=1;
    char ch;
    cin>>n>>ch;
    while(2*(k+1)*(k+1)-1<=n){
        k++;
    }
    int used=2*k*k-1;
    int remain=n-used;
    for(int i=k;i>=1;i--){
        for(int j=0;j<k-i;j++){
            cout<<' ';
        }
        for(int j=0;j<2*i-1;j++){
            cout<<ch;
        }
        cout<<endl;
    }
    for(int i=2;i<=k;i++){
        for(int j=0;j<k-i;j++){
            cout<<' ';
        }
        for(int j=0;j<2*i-1;j++){
            cout<<ch;
        }
        cout<<endl;
    }
    cout<<remain<<endl;
    return 0;
}
