#include<iostream>
#include<cstring>
using namespace std;

bool isong(const string& word){
    int len=word.length();
    if(len<3)return false;
    return word[len-3]=='o'&&word[len-2]=='n'&&word[len-1]=='g';
}
int main(){
    int n;
    cin>>n;
    cin.ignore();
    for(int i=0;i<n;i++){
        string line;
        getline(cin,line);
        int commapos=line.find(',');
        //将句子分为前后两句
        string firsthalf=line.substr(0,commapos);
        string secondhalf=line.substr(commapos+2);//跳过逗号
        //分别找到前后两句的最后一个词语
        int lastplace1=firsthalf.rfind(' ');//rfind从右往前找
        string lastword1=firsthalf.substr(lastplace1+1);
        int lastplace2=secondhalf.rfind(' ');
        string lastword2=secondhalf.substr(lastplace2+1,secondhalf.length()-lastplace2-2);//去掉句号
        if(isong(lastword1)&&isong(lastword2)){
            //找到后半句的最后三个词语进行替换
            int pos1=secondhalf.rfind(' ');
            int pos2=secondhalf.rfind(' ',pos1-1);
            int pos3=secondhalf.rfind(' ',pos2-1);
            string newsecondhalf=secondhalf.substr(0,pos3+1)+"qiao ben zhong.";
            cout<<firsthalf<<", "<<newsecondhalf<<endl;
        }else{
            cout<<"Skipped"<<endl;
        }
    }
    return 0;
}
