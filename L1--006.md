#include<iostream>
#include<cmath>
using namespace std;
int main(){
    int n;
    cin>>n;
    int maxlen=0;
    int start=0;
    for(int i=2;i<=sqrt(n);i++){
        int project=1;
        int j=i;
        while(true){
            project*=j;
            if(n%project!=0)break;
            if(j-i+1>maxlen){
                maxlen=j-i+1;
                start=i;
            }
            j++;
        }
    }
    if(maxlen==0)
        cout<<1<<endl<<n;
    else{
        cout<<maxlen<<endl;
        for(int i=0;i<maxlen;i++){
            if(i>0)cout<<"*";
            cout<<start+i;
        }
    }
    return 0;
}
