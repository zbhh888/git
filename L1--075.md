#include<iostream>
#include<iomanip>//补零头文件
using namespace std;
int main(){
    int  line;
    cin>>line;
    if(line/10000==0){
        int line1=line/100;
        int line2=line%100;
        //要记住补零，此时两个数字前都须补零
        if(line1<22)cout<<"20"<<setw(2)<<setfill('0')<<line1<<"-"<<setw(2)<<setfill('0')<<line2<<endl;
        else  cout<<"19"<<line1<<"-"<<setw(2)<<setfill('0')<<line2<<endl;
    }else{
        int line3=line/100;
        int line4=line%100;
        cout<<line3<<"-"<<setw(2)<<setfill('0')<<line4<<endl;
    }
    return 0;
}
