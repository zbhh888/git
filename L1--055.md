#include<iostream>
using namespace std;
int main(){
    int a,b;
    cin>>a>>b;
    int c[4];
    for(int i=0;i<3;i++){
        cin>>c[i];
    }
    int a1=0,b1=0;
    for(int i=0;i<3;i++){
        if(c[i]==0){
            a1++;
        }
        else{
            b1++;
        }
    }
    if((a>b&&a1>0)||(a<b&&a1==3)){
        cout<<"The winner is a: "<<a<<" + "<<a1<<endl;
    }else{
        cout<<"The winner is b: "<<b<<" + "<<b1<<endl;
    }
    return 0;
}
